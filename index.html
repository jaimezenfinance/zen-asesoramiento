import json, io, os from http.server import BaseHTTPRequestHandler import openpyxl from openpyxl import load_workbook from openpyxl.styles import Font, PatternFill import warnings warnings.filterwarnings("ignore") class handler(BaseHTTPRequestHandler): def do_OPTIONS(self): self.send_response(200) self._cors() self.end_headers() def do_POST(self): try: length = int(self.headers.get("Content-Length", 0)) data = json.loads(self.rfile.read(length)) edad = int(data.get("edad", 0)) total_ingresos = float(data.get("totalIngresos", 0)) total_deudas = float(data.get("totalDeudas", 0)) presupuesto = float(data.get("presupuesto", 0)) precio_vivienda = float(data.get("precioVivienda", 0)) entrada = float(data.get("entrada", 0)) nombre = data.get("nombre", "Asesoramiento") tmpl = os.path.join(os.path.dirname(__file__), "..", "plantilla.xlsx") wb = load_workbook(tmpl) sim = wb["Simulador"] pre = wb["Preasesoramiento"] BLACK = Font(name="Arial", bold=True, size=10, color="000000") BLUE = Font(name="Arial", bold=True, size=10, color="4472C4") ORANGE = PatternFill("solid", start_color="FFE699", end_color="FFE699") def s(sheet, coord, val, font): sheet[coord].value = val sheet[coord].font = font def p(coord, val): pre[coord].value = val pre[coord].font = BLUE pre[coord].fill = ORANGE s(sim, "A9", edad, BLACK) s(sim, "C9", total_ingresos, BLACK) s(sim, "D9", total_deudas, BLACK) s(sim, "B16", presupuesto, BLACK) if precio_vivienda > 0: p("B8", precio_vivienda) if entrada > 0: p("B9", entrada) p("B11", 0.035 if edad <= 35 else 0.06) buf = io.BytesIO() wb.save(buf) xlsx = buf.getvalue() fname = f"Zen_{nombre.replace(' ','_')}.xlsx" self.send_response(200) self._cors() self.send_header("Content-Type", "application/vnd.openxmlformats-officedocument.spreadsheetml.sheet") self.send_header("Content-Disposition", f'attachment; filename="{fname}"') self.send_header("Content-Length", str(len(xlsx))) self.end_headers() self.wfile.write(xlsx) except Exception as e: self.send_response(500) self._cors() self.send_header("Content-Type", "application/json") self.end_headers() self.wfile.write(json.dumps({"error": str(e)}).encode()) def _cors(self): self.send_header("Access-Control-Allow-Origin", "*") self.send_header("Access-Control-Allow-Methods", "POST, OPTIONS") self.send_header("Access-Control-Allow-Headers", "Content-Type")
